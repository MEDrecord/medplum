# Gateway Patterns

## 1. API Route Template

```typescript
// app/api/[resource]/route.ts
import { NextResponse } from 'next/server';
import { verifyGatewayAuth, GatewayAuthError } from '@/lib/gateway';

export async function GET(request: Request) {
  try {
    // 1. Auth verification - ALWAYS first
    const auth = await verifyGatewayAuth(request);
    
    // 2. Parse params
    const url = new URL(request.url);
    const param = url.searchParams.get('param');
    
    // 3. Business logic
    const data = await fetchData(auth.tenant_id);
    
    // 4. Return response
    return NextResponse.json({
      data,
      meta: { tenant_id: auth.tenant_id }
    });
    
  } catch (error) {
    return handleApiError(error);
  }
}

export async function POST(request: Request) {
  try {
    const auth = await verifyGatewayAuth(request);
    const body = await request.json();
    
    // Validate
    const validated = Schema.parse(body);
    
    // Create
    const created = await createResource(validated, auth);
    
    return NextResponse.json({ data: created }, { status: 201 });
    
  } catch (error) {
    return handleApiError(error);
  }
}

function handleApiError(error: unknown) {
  if (error instanceof GatewayAuthError) {
    return NextResponse.json(
      { error: { code: error.code, message: error.message } },
      { status: 401 }
    );
  }
  
  if (error instanceof z.ZodError) {
    return NextResponse.json(
      { error: { code: 'VALIDATION_ERROR', details: error.flatten() } },
      { status: 400 }
    );
  }
  
  console.error('[HealthTalk] API Error:', error);
  return NextResponse.json(
    { error: { code: 'INTERNAL_ERROR', message: 'Internal server error' } },
    { status: 500 }
  );
}
```

## 2. Detail Route Template

```typescript
// app/api/[resource]/[id]/route.ts
import { NextResponse } from 'next/server';
import { verifyGatewayAuth } from '@/lib/gateway';

interface RouteParams {
  params: Promise<{ id: string }>;
}

export async function GET(request: Request, { params }: RouteParams) {
  try {
    const auth = await verifyGatewayAuth(request);
    const { id } = await params;
    
    const resource = await getResourceById(id, auth.tenant_id);
    
    if (!resource) {
      return NextResponse.json(
        { error: { code: 'NOT_FOUND', message: 'Resource not found' } },
        { status: 404 }
      );
    }
    
    return NextResponse.json({ data: resource });
    
  } catch (error) {
    return handleApiError(error);
  }
}

export async function PUT(request: Request, { params }: RouteParams) {
  try {
    const auth = await verifyGatewayAuth(request);
    const { id } = await params;
    const body = await request.json();
    
    const updated = await updateResource(id, body, auth);
    
    return NextResponse.json({ data: updated });
    
  } catch (error) {
    return handleApiError(error);
  }
}

export async function DELETE(request: Request, { params }: RouteParams) {
  try {
    const auth = await verifyGatewayAuth(request);
    const { id } = await params;
    
    await deleteResource(id, auth.tenant_id);
    
    return new NextResponse(null, { status: 204 });
    
  } catch (error) {
    return handleApiError(error);
  }
}
```

## 3. Hook Pattern

```typescript
// hooks/use-[resource].ts
import useSWR from 'swr';
import { gatewayFetch } from '@/lib/api/gateway-fetch';

interface UseResourceOptions {
  enabled?: boolean;
}

export function useResource(id: string | undefined, options?: UseResourceOptions) {
  const { enabled = true } = options || {};
  
  const { data, error, isLoading, mutate } = useSWR(
    enabled && id ? ['resource', id] : null,
    async () => {
      const response = await gatewayFetch(`/api/resources/${id}`);
      if (!response.ok) {
        throw new Error('Failed to fetch resource');
      }
      return response.json();
    }
  );
  
  return {
    resource: data?.data,
    isLoading,
    error,
    refresh: mutate
  };
}
```

## 4. List Hook Pattern

```typescript
// hooks/use-[resources]-list.ts
import useSWR from 'swr';
import { gatewayFetch } from '@/lib/api/gateway-fetch';

interface ListOptions {
  page?: number;
  limit?: number;
  search?: string;
  status?: string;
}

export function useResourcesList(options?: ListOptions) {
  const params = new URLSearchParams();
  if (options?.page) params.set('page', String(options.page));
  if (options?.limit) params.set('limit', String(options.limit));
  if (options?.search) params.set('search', options.search);
  if (options?.status) params.set('status', options.status);
  
  const { data, error, isLoading, mutate } = useSWR(
    ['resources', params.toString()],
    async () => {
      const response = await gatewayFetch(`/api/resources?${params}`);
      if (!response.ok) {
        throw new Error('Failed to fetch resources');
      }
      return response.json();
    }
  );
  
  return {
    resources: data?.data || [],
    pagination: data?.meta?.pagination,
    isLoading,
    error,
    refresh: mutate
  };
}
```

## 5. Mutation Hook Pattern

```typescript
// hooks/use-create-[resource].ts
import { useCallback, useState } from 'react';
import { gatewayFetch } from '@/lib/api/gateway-fetch';

interface CreateResourceRequest {
  // ... fields
}

export function useCreateResource() {
  const [isCreating, setIsCreating] = useState(false);
  const [error, setError] = useState<Error | null>(null);
  
  const createResource = useCallback(async (
    request: CreateResourceRequest
  ) => {
    setIsCreating(true);
    setError(null);
    
    try {
      const response = await gatewayFetch('/api/resources', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify(request)
      });
      
      if (!response.ok) {
        const errorData = await response.json();
        throw new Error(errorData.error?.message || 'Failed to create');
      }
      
      return response.json();
      
    } catch (err) {
      setError(err as Error);
      throw err;
    } finally {
      setIsCreating(false);
    }
  }, []);
  
  return { createResource, isCreating, error };
}
```

## 6. Gateway Fetch Wrapper

```typescript
// lib/api/gateway-fetch.ts
export async function gatewayFetch(
  url: string,
  options: RequestInit = {}
): Promise<Response> {
  const baseUrl = process.env.NEXT_PUBLIC_GATEWAY_BASE_URL;
  const fullUrl = url.startsWith('/') ? `${baseUrl}${url}` : url;
  
  const headers = new Headers(options.headers);
  
  // Add auth
  const sessionId = typeof window !== 'undefined' 
    ? localStorage.getItem('sessionId') 
    : null;
  
  if (sessionId) {
    headers.set('X-Session-Id', sessionId);
  }
  
  // Add CSRF for mutations
  if (options.method && options.method !== 'GET') {
    const csrfToken = getCsrfToken();
    if (csrfToken) {
      headers.set('X-CSRF-Token', csrfToken);
    }
  }
  
  const response = await fetch(fullUrl, {
    ...options,
    headers,
    credentials: 'include'
  });
  
  // Handle auth errors globally
  if (response.status === 401 || response.status === 403) {
    window.dispatchEvent(new CustomEvent('gateway:auth-error', {
      detail: { status: response.status, url }
    }));
  }
  
  return response;
}
```
