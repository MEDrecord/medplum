# FHIR Skills

## 1. Questionnaire Creation

### Basic Questionnaire

```typescript
import type { Questionnaire } from '@medplum/fhirtypes';

const questionnaire: Questionnaire = {
  resourceType: 'Questionnaire',
  status: 'draft',
  title: 'Patient Intake',
  description: 'Initial patient intake form',
  item: [
    {
      linkId: 'demographics',
      text: 'Demographics',
      type: 'group',
      item: [
        {
          linkId: 'name',
          text: 'Full Name',
          type: 'string',
          required: true
        },
        {
          linkId: 'dob',
          text: 'Date of Birth',
          type: 'date',
          required: true
        }
      ]
    }
  ]
};
```

### ROM Questionnaire with Scoring

```typescript
const phq9: Questionnaire = {

        { valueCoding: { code: '1', display: 'Several days' } },
        { valueCoding: { code: '2', display: 'More than half the days' } },
        { valueCoding: { code: '3', display: 'Nearly every day' } }
      ]
    },
    // ... q2-q9
    {
      linkId: 'total-score',
      text: 'Total Score',
      type: 'integer',
      readOnly: true,
      extension: [{
        url: 'http://hl7.org/fhir/uv/sdc/StructureDefinition/sdc-questionnaire-calculatedExpression',
        valueExpression: {
          language: 'text/fhirpath',
          expression: "item.where(linkId.startsWith('q')).answer.valueCoding.code.aggregate($this.toInteger() + $total, 0)"
        }
      }]
    }
  ]
};
```

## 3. Score Observation Creation

```typescript
import type { Observation } from '@medplum/fhirtypes';

function createScoreObservation(
  score: number,
  patientId: string,
  responseId: string,
  interpretation: { code: string; display: string }
): Observation {
  return {
    resourceType: 'Observation',
    status: 'final',
    category: [{
      coding: [{
        system: 'http://terminology.hl7.org/CodeSystem/observation-category',
        code: 'survey',
        display: 'Survey'
      }]
    }],
    code: {
      coding: [{
        system: 'http://loinc.org',
        code: '44261-6',
        display: 'PHQ-9 total score'
      }]
    },
    subject: { reference: `Patient/${patientId}` },
    effectiveDateTime: new Date().toISOString(),
    valueInteger: score,
    derivedFrom: [{ reference: `QuestionnaireResponse/${responseId}` }],
    interpretation: [{
      coding: [{
        system: 'https://medrecord.nl/fhir/score-severity',
        code: interpretation.code,
        display: interpretation.display
      }]
    }]
  };
}
```

## 4. FHIRPath Expressions

### Sum All Choice Answers

```fhirpath
item.where(linkId.startsWith('q')).answer.valueCoding.code.aggregate($this.toInteger() + $total, 0)
```

### Count Positive Answers

```fhirpath
item.where(linkId.startsWith('q')).answer.valueCoding.code.where($this.toInteger() > 0).count()
```

### Conditional Display

```fhirpath
%resource.item.where(linkId='q9').answer.valueCoding.code.toInteger() > 0
```

## 6. Search Operations

### Search Questionnaires

```typescript
const bundle = await medplum.search('Questionnaire', {
  status: 'active',
  title: 'PHQ',
  _sort: '-_lastUpdated',
  _count: '50'
});
```

### Search Responses for Patient

```typescript
const bundle = await medplum.search('QuestionnaireResponse', {
  subject: `Patient/${patientId}`,
  questionnaire: `Questionnaire/${questionnaireId}`,
  _sort: '-authored'
});
```

### Search with Include

```typescript
const bundle = await medplum.search('QuestionnaireResponse', {
  subject: `Patient/${patientId}`,
  _include: 'QuestionnaireResponse:questionnaire'
});
```

## 8. LOINC Integration

### Fetch from LOINC

```typescript
async function fetchLoincQuestionnaire(code: string): Promise<Questionnaire> {
  const response = await fetch(
    `https://fhir.loinc.org/Questionnaire/?url=http://loinc.org/q/${code}`,
    {
      headers: {
        Authorization: `Basic ${btoa(`${LOINC_USERNAME}:${LOINC_PASSWORD}`)}`
      }
    }
  );
  
  const bundle = await response.json();
  const questionnaire = bundle.entry?.[0]?.resource;
  
  if (!questionnaire) {
    throw new Error('Questionnaire not found in LOINC');
  }
  
  // Remove ID and set as draft
  delete questionnaire.id;
  questionnaire.status = 'draft';
  
  // Add copyright
  questionnaire.copyright = `This content from LOINC® is copyright © 1995 Regenstrief Institute, Inc.
and the LOINC Committee, and available at no cost under the license at
http://loinc.org/terms-of-use`;
  
  return questionnaire;
}
```
