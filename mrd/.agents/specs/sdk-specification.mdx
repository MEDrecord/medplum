# @mrd/sdk Specification

The SDK provides a type-safe client and React hooks for all brand applications to communicate with the Gateway service.

## Installation

```bash
pnpm add @mrd/sdk @mrd/shared
```

## GatewayClient

### Initialization

```typescript
import { GatewayClient } from '@mrd/sdk';

const client = new GatewayClient({
  gatewayUrl: 'https://gateway.medrecord.nl',
  brand: 'healthtalk',
  tenantId: 'tenant-123',
  getAccessToken: async () => {
    // Return current session token
    return getSessionToken();
  },
});
```

### Configuration Options

| Option | Type | Required | Description |
|--------|------|----------|-------------|
| `gatewayUrl` | `string` | Yes | Gateway service URL |
| `brand` | `BrandId` | Yes | Brand identifier |
| `tenantId` | `string` | No | Tenant ID (from session) |
| `getAccessToken` | `() => Promise<string>` | Yes | Token provider function |

### API Methods

#### Templates

```typescript
// List templates
const { templates, pagination } = await client.templates.list({
  type: 'questionnaire',
  status: 'published',
  page: 1,
  limit: 20,
});

// Get single template
const template = await client.templates.get('template-123');

// Create template
const newTemplate = await client.templates.create({
  title: 'PHQ-9',
  type: 'questionnaire',
  language: 'nl',
  category: 'mental-health',
});

// Update template
const updated = await client.templates.update('template-123', {
  title: 'PHQ-9 Updated',
});

// Delete template
await client.templates.delete('template-123');

// Publish template
await client.templates.publish('template-123');
```

#### Sections (Note Templates)

```typescript
// Get sections for a template
const sections = await client.templates.getSections('template-123');

// Add section
const section = await client.templates.addSection('template-123', {
  title: 'Subjective',
  prompt: 'Summarize the patient complaints...',
  order: 1,
});

// Update section
await client.templates.updateSection('template-123', 'section-456', {
  prompt: 'Updated prompt...',
});

// Reorder sections
await client.templates.reorderSections('template-123', [
  'section-1',
  'section-3',
  'section-2',
]);
```

#### Questionnaires

```typescript
// Get questionnaire (FHIR Questionnaire resource)
const questionnaire = await client.questionnaires.get('questionnaire-123');

// Submit response
const response = await client.questionnaires.submitResponse('questionnaire-123', {
  subject: { reference: 'Patient/456' },
  encounter: { reference: 'Encounter/789' },
  item: [
    { linkId: 'q1', answer: [{ valueInteger: 2 }] },
    { linkId: 'q2', answer: [{ valueInteger: 1 }] },
  ],
});
```

#### Appointments

```typescript
// List appointments
const appointments = await client.appointments.list({
  date: '2026-04-06',
  practitioner: 'Practitioner/123',
});

// Get appointment with patient info
const appointment = await client.appointments.get('appointment-123');
```

#### Notifications

```typescript
// Send questionnaire via best channel
await client.notifications.send({

// Send via specific channel
await client.notifications.send({

```


## React Integration

### GatewayProvider

Wrap your app with `GatewayProvider` to enable hooks:

```typescript
// app/layout.tsx
import { GatewayProvider } from '@mrd/sdk';

export default function RootLayout({ children }) {
  return (
    <GatewayProvider
      gatewayUrl={process.env.NEXT_PUBLIC_GATEWAY_URL!}
      brand="healthtalk"
      tenantId={session.tenantId}
      userId={session.userId}
      getAccessToken={() => getSessionToken()}
    >
      {children}
    </GatewayProvider>
  );
}
```

### useGateway

Access the client and context in any component:

```typescript
import { useGateway } from '@mrd/sdk';

function MyComponent() {
  const { client, brand, tenantId, userId } = useGateway();
  
  // Use client directly for custom operations
  const handleCustomAction = async () => {
    await client.templates.publish('template-123');
  };
}
```

### useTemplatesList

```typescript
import { useTemplatesList } from '@mrd/sdk';

function TemplateList() {
  const { data, error, isLoading, mutate } = useTemplatesList({
    type: 'questionnaire',
    status: 'published',
  });

  if (isLoading) return <Skeleton />;
  if (error) return <Error message={error.message} />;

  return (
    <ul>
      {data.templates.map(template => (
        <li key={template.id}>{template.title}</li>
      ))}
    </ul>
  );
}
```

### useTemplateById

```typescript
import { useTemplateById } from '@mrd/sdk';

function TemplateDetail({ id }: { id: string }) {
  const { data: template, error, isLoading } = useTemplateById(id);

  if (isLoading) return <Skeleton />;
  if (error) return <Error message={error.message} />;

  return (
    <div>
      <h1>{template.title}</h1>
      {template.type === 'note-template' && (
        <SectionEditor sections={template.sections} />
      )}
      {template.type === 'questionnaire' && (
        <QuestionnaireBuilder questionnaire={template.questionnaire} />
      )}
    </div>
  );
}
```

### useQuestionnaireForm

Headless hook for questionnaire form state:

```typescript
import { useQuestionnaireForm } from '@mrd/sdk';
import type { Questionnaire } from '@medplum/fhirtypes';

function PatientQuestionnaireForm({ questionnaire }: { questionnaire: Questionnaire }) {
  const form = useQuestionnaireForm({
    questionnaire,
    language: 'nl',
    subject: { reference: 'Patient/123' },
  });

  return (
    <form onSubmit={() => form.submit()}>
      {form.items.map(item => (
        <QuestionRenderer
          key={item.linkId}
          item={item}
          value={form.getValue(item.linkId)}
          onChange={(value) => form.setValue(item.linkId, value)}
          patientText={form.getPatientText(item)}
          error={form.getError(item.linkId)}
        />
      ))}

      <div className="flex justify-between">
        {form.hasPrevPage && (
          <Button onClick={form.prevPage}>Previous</Button>
        )}
        {form.hasNextPage ? (
          <Button onClick={form.nextPage}>Next</Button>
        ) : (
          <Button type="submit" disabled={!form.isValid}>
            Submit
          </Button>
        )}
      </div>
    </form>
  );
}
```

#### useQuestionnaireForm API

| Property/Method | Type | Description |
|-----------------|------|-------------|
| `items` | `QuestionnaireItem[]` | Items for current page |
| `currentPage` | `number` | Current page index |
| `totalPages` | `number` | Total number of pages |
| `hasPrevPage` | `boolean` | Can go to previous page |
| `hasNextPage` | `boolean` | Can go to next page |
| `isValid` | `boolean` | All required fields filled |
| `getValue(linkId)` | `any` | Get answer value |
| `setValue(linkId, value)` | `void` | Set answer value |
| `getPatientText(item)` | `string` | Get patient-friendly text |
| `getError(linkId)` | `string \| null` | Get validation error |
| `prevPage()` | `void` | Go to previous page |
| `nextPage()` | `void` | Go to next page |
| `submit()` | `Promise<QuestionnaireResponse>` | Submit to Gateway |
| `buildResponse()` | `QuestionnaireResponse` | Build response without submitting |

### useAuth

```typescript
import { useAuth } from '@mrd/sdk';

function UserMenu() {
  const { user, isAuthenticated, login, logout } = useAuth();

  if (!isAuthenticated) {
    return <Button onClick={login}>Login</Button>;
  }

  return (
    <DropdownMenu>
      <span>{user.name}</span>
      <Button onClick={logout}>Logout</Button>
    </DropdownMenu>
  );
}
```

## Utilities

### Patient Language Helpers

```typescript
import { getPatientText, getPatientAnswerText } from '@mrd/sdk';

// Get patient-friendly question text
const questionText = getPatientText(item, 'nl');
// Falls back to item.text if no patient-question extension

// Get patient-friendly answer text
const answerText = getPatientAnswerText(item, answerOption, 'nl');
// Falls back to answerOption.valueCoding.display
```

### FHIR Helpers

```typescript
import { isChoiceQuestion, isRequiredQuestion, evaluateEnableWhen } from '@mrd/sdk';

// Check question type
if (isChoiceQuestion(item)) {
  // Render as radio or select
}

// Check if required
if (isRequiredQuestion(item)) {
  // Show required indicator
}

// Check if question should be shown
if (evaluateEnableWhen(item, response)) {
  // Render the question
}
```

## Error Handling

```typescript
import { GatewayError, isGatewayError } from '@mrd/sdk';

try {
  await client.templates.create({ ... });
} catch (error) {
  if (isGatewayError(error)) {
    switch (error.status) {
      case 401:
        // Redirect to login
        break;
      case 403:
        // Show permission denied
        break;
      case 422:
        // Show validation errors
        console.log(error.errors);
        break;
      default:
        // Show generic error
    }
  }
}
```
