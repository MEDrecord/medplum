### 2.3 Tenant Detectie

De tenant wordt bepaald via:

1. **Subdomain**: `tenant-slug.healthtalk.ai`
2. **Auth token**: `tenant_id` in JWT claims
3. **Header**: `X-Tenant-Id` voor API calls

```typescript
function getTenantFromRequest(request: Request): string {
  // 1. Check header
  const headerTenant = request.headers.get('X-Tenant-Id');
  if (headerTenant) return headerTenant;
  
  // 2. Check subdomain
  const host = request.headers.get('host');
  const subdomain = host?.split('.')[0];
  if (subdomain && subdomain !== 'www') {
    return lookupTenantBySlug(subdomain);
  }
  
  // 3. Fallback to auth
  const auth = await verifyGatewayAuth(request);
  return auth.tenant_id;
}
```

### 3.2 Brand Provider

```typescript
// components/brand-provider.tsx
export function BrandProvider({ 
  brand, 
  children 
}: { 
  brand: BrandId; 
  children: React.ReactNode 
}) {
  const config = brands[brand];
  
  return (
    <div data-brand={brand}>
      <BrandContext.Provider value={config}>
        {children}
      </BrandContext.Provider>
    </div>
  );
}

// Usage
export function useBrand(): BrandConfig {
  const context = useContext(BrandContext);
  if (!context) {
    throw new Error('useBrand must be used within BrandProvider');
  }
  return context;
}
```

### 3.3 Logo Component

export function BrandLogo({ 
  variant = 'light',
  className

  );

## 5. Multi-Tenant Data Isolation

### 5.1 Query Scoping

Alle queries worden automatisch gefilterd op `tenant_id`:

```typescript
// Medplum search with tenant scope
async function searchResources<T extends Resource>(
  resourceType: string,
  params: Record<string, string>
): Promise<T[]> {
  const auth = await getAuth();
  
  // Tenant scope wordt automatisch toegevoegd door Medplum Project
  const bundle = await medplum.search(resourceType, {
    ...params,
    _project: auth.tenant_id  // Medplum Project = Tenant
  });
  
  return bundle.entry?.map(e => e.resource as T) || [];
}
```

### 5.2 Medplum Project Mapping

| Concept | Medplum | HealthTalk |
|---------|---------|------------|
| Tenant | Project | Organization |
| User | Practitioner | User |
| Patient | Patient | Patient |
