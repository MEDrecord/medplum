# SDK Hooks Pattern

React hook patterns for @mrd/sdk. All hooks use SWR for caching and revalidation.

## Data Fetching Hooks

### Basic Usage

```typescript
import { useTemplatesList } from '@mrd/sdk';

function TemplateList() {
  const { data, error, isLoading, isValidating, mutate } = useTemplatesList();

  // isLoading: true on initial load
  // isValidating: true during revalidation (refresh)
  // mutate: function to trigger revalidation

  if (isLoading) return <Skeleton />;
  if (error) return <Error />;

  return <List items={data.templates} />;
}
```

### With Parameters

```typescript
function FilteredTemplates({ type, status }: FilterProps) {
  const { data } = useTemplatesList({
    type,      // 'note-template' | 'questionnaire'
    status,    // 'draft' | 'published'
    page: 1,
    limit: 20,
  });
}
```

### Conditional Fetching

```typescript
function TemplateDetail({ id }: { id: string | null }) {
  // Only fetch when id is provided
  const { data } = useTemplateById(id ?? undefined);

  // Or using SWR's conditional fetching
  const { data } = useTemplateById(id || null);
}
```

## Mutation Patterns

### Optimistic Updates

```typescript
import { useTemplatesList, useGateway } from '@mrd/sdk';
import { mutate } from 'swr';

function TemplateActions({ template }: { template: TemplateSummary }) {
  const { client } = useGateway();

  const handleDelete = async () => {
    // Optimistic update: remove from list immediately
    mutate(
      ['templates', { type: 'questionnaire' }],
      (current) => ({
        ...current,
        templates: current.templates.filter(t => t.id !== template.id),
      }),
      { revalidate: false }
    );

    try {
      await client.templates.delete(template.id);
      // Revalidate to get fresh data
      mutate(['templates', { type: 'questionnaire' }]);
    } catch (error) {
      // Revert on error
      mutate(['templates', { type: 'questionnaire' }]);
      toast.error('Failed to delete');
    }
  };
}
```

### After Mutation Revalidation

```typescript
function CreateTemplateForm() {
  const { client } = useGateway();
  const router = useRouter();

  const handleSubmit = async (data: CreateTemplateRequest) => {
    const template = await client.templates.create(data);
    
    // Revalidate the list
    mutate(['templates']);
    
    // Navigate to detail
    router.push(`/sjablonen/${template.id}`);
  };
}
```

## Form State Hooks

### useQuestionnaireForm

```typescript
import { useQuestionnaireForm } from '@mrd/sdk';

function PatientForm({ questionnaire }: Props) {
  const form = useQuestionnaireForm({
    questionnaire,
    language: 'nl',
  });

  return (
    <div>
      {/* Progress indicator */}
      <Progress value={(form.currentPage / form.totalPages) * 100} />

      {/* Current page items */}
      {form.items.map(item => (
        <QuestionField
          key={item.linkId}
          item={item}
          value={form.getValue(item.linkId)}
          onChange={(v) => form.setValue(item.linkId, v)}
          label={form.getPatientText(item)}
          required={item.required}
          error={form.getError(item.linkId)}
        />
      ))}

      {/* Navigation */}
      <div className="flex justify-between">
        <Button 
          onClick={form.prevPage} 
          disabled={!form.hasPrevPage}
        >
          Previous
        </Button>
        
        {form.hasNextPage ? (
          <Button onClick={form.nextPage}>
            Next
          </Button>
        ) : (
          <Button 
            onClick={form.submit}
            disabled={!form.isValid}
          >
            Submit
          </Button>
        )}
      </div>
    </div>
  );
}
```

### Question Type Rendering

```typescript
function QuestionField({ item, value, onChange, label }: Props) {
  switch (item.type) {
    case 'boolean':
      return (
        <RadioGroup value={value} onValueChange={onChange}>
          <RadioGroupItem value="true" label="Yes" />
          <RadioGroupItem value="false" label="No" />
        </RadioGroup>
      );

    case 'choice':
      if (item.answerOption.length <= 4) {
        return (
          <RadioGroup value={value} onValueChange={onChange}>
            {item.answerOption.map(opt => (
              <RadioGroupItem
                key={opt.valueCoding.code}
                value={opt.valueCoding.code}
                label={getPatientAnswerText(item, opt, 'nl')}
              />
            ))}
          </RadioGroup>
        );
      }
      return (
        <Select value={value} onValueChange={onChange}>
          {item.answerOption.map(opt => (
            <SelectItem
              key={opt.valueCoding.code}
              value={opt.valueCoding.code}
            >
              {getPatientAnswerText(item, opt, 'nl')}
            </SelectItem>
          ))}
        </Select>
      );

    case 'integer':
      return (
        <Input
          type="number"
          value={value}
          onChange={e => onChange(parseInt(e.target.value))}
        />
      );

    case 'string':
      return (
        <Textarea
          value={value}
          onChange={e => onChange(e.target.value)}
          maxLength={item.maxLength}
        />
      );

    case 'display':
      return <p className="text-muted-foreground">{label}</p>;

    default:
      return null;
  }
}
```

## Polling Pattern

For long-running operations like AI generation:

```typescript
import { useCallback, useEffect, useState } from 'react';
import { useGateway } from '@mrd/sdk';

function useGenerationStatus(generationId: string | null) {
  const { client } = useGateway();
  const [status, setStatus] = useState<'pending' | 'completed' | 'failed'>('pending');
  const [result, setResult] = useState(null);

  useEffect(() => {
    if (!generationId) return;

    const poll = async () => {
      const { status, result } = await client.ai.getStatus(generationId);
      setStatus(status);
      
      if (status === 'completed') {
        setResult(result);
        return true; // Stop polling
      }
      if (status === 'failed') {
        return true; // Stop polling
      }
      return false; // Continue polling
    };

    const interval = setInterval(async () => {
      const done = await poll();
      if (done) clearInterval(interval);
    }, 2000);

    return () => clearInterval(interval);
  }, [generationId, client]);

  return { status, result };
}
```

## Hook Composition

Combine hooks for complex views:

```typescript
function AppointmentWithQuestionnaire({ appointmentId }: Props) {
  // Load appointment
  const { data: appointment } = useAppointmentById(appointmentId);
  
  // Load linked questionnaire task
  const { data: task } = useTaskByAppointment(
    appointment?.id ?? null
  );
  
  // Load questionnaire if task exists
  const { data: questionnaire } = useQuestionnaireById(
    task?.focus?.reference?.split('/')[1] ?? null
  );

  // All loading states
  const isLoading = !appointment || (task && !questionnaire);

  if (isLoading) return <Skeleton />;

  return (
    <div>
      <AppointmentCard appointment={appointment} />
      {questionnaire && (
        <QuestionnairePreview questionnaire={questionnaire} />
      )}
    </div>
  );
}
```
