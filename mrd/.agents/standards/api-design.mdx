# API Design Standard

Standards for designing and implementing Gateway API routes.

## URL Structure

```
/api/{resource}                    # Collection
/api/{resource}/{id}               # Single resource
/api/{resource}/{id}/{subresource} # Nested resource
```

### Examples

```
GET    /api/templates              # List templates
POST   /api/templates              # Create template
GET    /api/templates/123          # Get template
PATCH  /api/templates/123          # Update template
DELETE /api/templates/123          # Delete template
GET    /api/templates/123/sections # List sections
POST   /api/templates/123/sections # Add section
```

## HTTP Methods

| Method | Usage | Idempotent |
|--------|-------|------------|
| `GET` | Retrieve resource(s) | Yes |
| `POST` | Create new resource | No |
| `PATCH` | Partial update | Yes |
| `DELETE` | Remove resource | Yes |

**Never use PUT** — always use PATCH for updates.

## Request Headers

### Required Headers

| Header | Description | Example |
|--------|-------------|---------|
| `Content-Type` | Request body format | `application/json` |
| `X-Brand` | Brand identifier | `healthtalk` |

### Authentication

Session-based via HTTP-only cookie:

```
Cookie: session=eyJhbGciOiJIUzI1NiIs...
```

## Response Format

### Success Response

```typescript
// Single resource
{
  "id": "template-123",
  "title": "PHQ-9",
  "type": "questionnaire",
  ...
}

// Collection
{
  "templates": [...],
  "pagination": {
    "page": 1,
    "limit": 20,
    "total": 45,
    "totalPages": 3
  }
}
```

### Error Response

```typescript
{
  "error": {
    "code": "VALIDATION_ERROR",
    "message": "Invalid request data",
    "errors": [
      {
        "field": "title",
        "message": "Title is required"
      }
    ]
  }
}
```

## HTTP Status Codes

| Code | Usage |
|------|-------|
| `200` | Success (GET, PATCH) |
| `201` | Created (POST) |
| `204` | No content (DELETE) |
| `400` | Bad request (invalid syntax) |
| `401` | Unauthorized (no/invalid session) |
| `403` | Forbidden (no permission) |
| `404` | Not found |
| `422` | Validation error |
| `500` | Server error |

## Query Parameters

### Pagination

```
GET /api/templates?page=2&limit=20
```

| Parameter | Type | Default | Max |
|-----------|------|---------|-----|
| `page` | number | 1 | - |
| `limit` | number | 20 | 100 |

### Filtering

```
GET /api/templates?type=questionnaire&status=published
```

Use exact match for enum fields.

### Sorting

```
GET /api/templates?sort=createdAt&order=desc
```

| Parameter | Values |
|-----------|--------|
| `sort` | Field name |
| `order` | `asc` \| `desc` |

### Search

```
GET /api/templates?search=PHQ
```

Searches across relevant text fields (title, description).

## Route Handler Pattern

```typescript
// app/api/templates/route.ts
import { NextRequest, NextResponse } from 'next/server';
import { authMiddleware } from '@/lib/middleware/auth-middleware';
import { brandMiddleware } from '@/lib/middleware/brand-middleware';
import { getMedplumClient } from '@/lib/medplum';
import { z } from 'zod';

// Validation schema
const CreateTemplateSchema = z.object({
  title: z.string().min(1).max(200),
  type: z.enum(['note-template', 'questionnaire']),
  language: z.string().default('nl'),
  category: z.string().optional(),
});

export async function GET(request: NextRequest) {
  try {
    // Authenticate
    const auth = await authMiddleware(request);
    const brand = brandMiddleware(request);
    
    // Parse query params
    const { searchParams } = new URL(request.url);
    const page = parseInt(searchParams.get('page') || '1');
    const limit = Math.min(parseInt(searchParams.get('limit') || '20'), 100);
    const type = searchParams.get('type');
    const status = searchParams.get('status');
    
    // Fetch from MedPlum
    const medplum = getMedplumClient();
    medplum.setProject(auth.tenantId);
    
    const templates = await medplum.searchResources('PlanDefinition', {
      _count: limit,
      _offset: (page - 1) * limit,
      ...(type && { type }),
      ...(status && { status }),
    });
    
    const total = await medplum.searchCount('PlanDefinition', {
      ...(type && { type }),
      ...(status && { status }),
    });
    
    return NextResponse.json({
      templates: templates.map(mapToTemplateSummary),
      pagination: {
        page,
        limit,
        total,
        totalPages: Math.ceil(total / limit),
      },
    });
  } catch (error) {
    return handleError(error);
  }
}

export async function POST(request: NextRequest) {
  try {
    const auth = await authMiddleware(request);
    const brand = brandMiddleware(request);
    
    // Parse and validate body
    const body = await request.json();
    const data = CreateTemplateSchema.parse(body);
    
    // Create in MedPlum
    const medplum = getMedplumClient();
    medplum.setProject(auth.tenantId);
    
    const template = await medplum.createResource({
      resourceType: 'PlanDefinition',
      title: data.title,
      status: 'draft',
      // ...
    });
    
    return NextResponse.json(
      mapToTemplateDetail(template),
      { status: 201 }
    );
  } catch (error) {
    return handleError(error);
  }
}
```

## Error Handling

```typescript
// lib/errors.ts
export class ApiError extends Error {
  constructor(
    public status: number,
    public code: string,
    message: string,
    public errors?: { field: string; message: string }[]
  ) {
    super(message);
  }
}

// lib/handle-error.ts
import { NextResponse } from 'next/server';
import { ZodError } from 'zod';

export function handleError(error: unknown): NextResponse {
  // Validation error
  if (error instanceof ZodError) {
    return NextResponse.json({
      error: {
        code: 'VALIDATION_ERROR',
        message: 'Invalid request data',
        errors: error.errors.map(e => ({
          field: e.path.join('.'),
          message: e.message,
        })),
      },
    }, { status: 422 });
  }

  // Known API error
  if (error instanceof ApiError) {
    return NextResponse.json({
      error: {
        code: error.code,
        message: error.message,
        errors: error.errors,
      },
    }, { status: error.status });
  }

  // Unknown error
  console.error('Unhandled error:', error);
  return NextResponse.json({
    error: {
      code: 'INTERNAL_ERROR',
      message: 'An unexpected error occurred',
    },
  }, { status: 500 });
}
```

## Authentication Middleware

```typescript
// lib/middleware/auth-middleware.ts
import { cookies } from 'next/headers';
import { verifySession } from '../auth';
import { ApiError } from '../errors';

export interface AuthContext {
  userId: string;
  tenantId: string;
  brand: string;
  roles: string[];
}

export async function authMiddleware(request: Request): Promise<AuthContext> {
  const cookieStore = await cookies();
  const sessionCookie = cookieStore.get('session');

  if (!sessionCookie) {
    throw new ApiError(401, 'UNAUTHORIZED', 'No session found');
  }

  try {
    const session = await verifySession(sessionCookie.value);
    return {
      userId: session.userId,
      tenantId: session.tenantId,
      brand: session.brand,
      roles: session.roles,
    };
  } catch {
    throw new ApiError(401, 'UNAUTHORIZED', 'Invalid session');
  }
}
```

## Rate Limiting

Apply rate limits per tenant:

```typescript
// lib/rate-limit.ts
import { Ratelimit } from '@upstash/ratelimit';
import { Redis } from '@upstash/redis';

const ratelimit = new Ratelimit({
  redis: Redis.fromEnv(),
  limiter: Ratelimit.slidingWindow(100, '1 m'),
});

export async function checkRateLimit(tenantId: string): Promise<void> {
  const { success } = await ratelimit.limit(tenantId);
  if (!success) {
    throw new ApiError(429, 'RATE_LIMITED', 'Too many requests');
  }
}
```

  method: string,

    path,

    brand: auth.brand,
    status,
    durationMs,
  }));
}
